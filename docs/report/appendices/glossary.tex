%TC:ignore
\newterm{llm}{Large language model (LLM)}{A neural network model pre-trained on large amounts of text, here fine-tuned to behave in line with survey responses.}
\newterm{ai-system}{AI system}{A computer system that uses one or many forms of artificial intelligence to perform tasks that would normally require human intelligence.}
\newterm{agentic-harness}{Agentic harness}{A framework that gives a model (normally a LLM) the abilityt o act autonomousy by allows its outputs to be used as inputs to other systems (like a search engine, online shopping, code editor etc). }

\newterm{value-survey}{Value survey}{A survey that asks a person to answer questions about their values and preferences. Common examples globally are the World Values Survey and the Pew Research Center surveys.}
