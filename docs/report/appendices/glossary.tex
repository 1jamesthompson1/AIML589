%TC:ignore
\newterm{llm}{large language model (LLM)}{large language models (LLMs)}{A neural network model pre-trained on large amounts of text and trained to predict the next token.}
\newterm{ai-system}{AI system}{AI systems}{A computer system that performs tasks that would normally require human intelligence.}
\newterm{agent}{agent}{agents}{A system that can perceive its environment and take actions to achieve a goal. In this report, an AI agent is a model wrapped in a harness that gives it a goal and tools.}
\newterm{agentic-harness}{Agentic harness}{Agentic harnesses}{A framework that gives a model a goal and tools, and lets its outputs be used as inputs to other systems (such as a search engine, an online shop or a code editor).}

\newterm{value-survey}{Value survey}{Value surveys}{A survey that asks a person to answer questions about their values and preferences. Common examples globally are the World Values Survey and the Pew Research Center surveys.}

\newterm{work-profile}{Work profile}{Work profiles}{The system prompt that places an AI agent in a deployment context. A work profile is paired with a situation to form a dilemma.}

\newterm{dilemma}{Dilemma}{Dilemmas}{A situation with no clear right answer, where a trade-off between competing values must be made.}

\newterm{value-alignment}{Value alignment}{Value alignments}{The state in which an AI system behaves in accordance with the values of a specified population.}

\newterm{open-weight}{open-weight}{open-weight}{A model whose trained parameters are publicly released, so that others can run or fine-tune it.}

\newterm{open-source}{open-source}{open-source}{A model whose weights, code and training details are released under an open licence.}

\newterm{value-action-gap}{Value-action gap}{Value-action gaps}{The gap between what people say they value and how they act.}
