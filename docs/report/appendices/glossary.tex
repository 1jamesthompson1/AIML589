%TC:ignore
\newterm{llm}{Large language model (LLM)}{Large language models (LLMs)}{A neural network model pre-trained on large amounts of text, here fine-tuned to behave in line with survey responses.}
\newterm{ai-system}{AI system}{AI systems}{A computer system that uses one or many forms of artificial intelligence to perform tasks that would normally require human intelligence.}
\newterm{agent}{agent}{agents}{"A person or thing that causes something to happen" \cite{DefinitionAGENT2026}. The key framework is that an agent has a goal and can take actions that have an effect on the world to achieve that goal. The world in this case could be the physical world, the digital world, or a simulated environment.}
\newterm{agentic-harness}{Agentic harness}{Agentic harnesses}{A framework that gives a model (normally a LLM) the ability to act autonomousy by allows its outputs to be used as inputs to other systems (like a search engine, online shopping, code editor etc). }

\newterm{value-survey}{Value survey}{Value surveys}{A survey that asks a person to answer questions about their values and preferences. Common examples globally are the World Values Survey and the Pew Research Center surveys.}

\newterm{work-profile}{Work profile}{Work profiles}{An envisioned deployment scenario that an 
AI agent may be used in.}

\newterm{dilemma}{Dilemma}{Dilemmas}{"A usually undesirable or unpleasant choice" \cite{DefinitionDILEMMA2026}.}

\newterm{value-alignment}{Value alignment}{Value alignments}{The state of an AI system holding that are aligned with the values set by the developers}
