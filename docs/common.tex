% Figure convenience: \cfigure[placement][width]{filename}{caption}
% The optional placement defaults to [t] (top of the page) and the caption is
% placed below the image.
% The label is auto-derived from the filename without extension, e.g.
% \cfigure{bootstrap_stability.png}{Caption text} -> \label{fig:bootstrap_stability}
% Images are loaded from the repo's code/figures/ dir (requires the xstring package)
% The optional width defaults to 0.8\textwidth; use [\textwidth] for full width.
% Strip a known file extension for use as a label; if there is no known
% extension (e.g. a bare stem like "bootstrap_stability", or a name containing
% dots such as a model version "ft-results-Qwen3.5-9B") the whole name is used.
% This keeps labels unique: stripping at the first dot would map every
% Qwen3.x table to the same label.
\newcommand{\stripext}[2]{%
   \def#2{#1}%
   \IfEndWith{#1}{.tex}{\StrBefore{#1}{.tex}[#2]}{}%
   \IfEndWith{#1}{.png}{\StrBefore{#1}{.png}[#2]}{}%
   \IfEndWith{#1}{.pdf}{\StrBefore{#1}{.pdf}[#2]}{}%
   \IfEndWith{#1}{.jpg}{\StrBefore{#1}{.jpg}[#2]}{}%
   \IfEndWith{#1}{.jpeg}{\StrBefore{#1}{.jpeg}[#2]}{}%
}

\NewDocumentCommand{\cfigure}{O{t} O{0.8\textwidth} m m}{%
   \stripext{#3}{\cfiglabel}%
   \begin{figure}[#1]
      \centering
      \includegraphics[width=#2]{../../code/figures/#3}
      \caption{#4}
      \label{fig:\cfiglabel}
   \end{figure}%
}

% Diagram convenience: \cdiagram[placement]{name}{caption}
% Inputs the bare TikZ picture diagrams/<name>.tex (relative to docs/report, no preamble) inside a figure
% environment. The label is auto-derived from the name, e.g. \cdiagram{foo}{...} gives
% \label{fig:foo}, so \figref{foo} works. Pictures are drawn at \textwidth.
\NewDocumentCommand{\cdiagram}{O{t} m m}{%
   \stripext{#2}{\cdiagname}%
   \begin{figure}[#1]
      \centering
      \input{diagrams/\cdiagname.tex}
      \caption{#3}
      \label{fig:\cdiagname}
   \end{figure}%
}

% Figure reference convenience: \figref{filename} -> "Figure N"
% A comma list works too: \figref{a.png,b.png} -> "Figures 3.1 and 3.2".
% Uses the same filename-derived label as \cfigure, e.g.
% \cfigure{bootstrap_stability.png}{...} then \figref{bootstrap_stability.png}
\makeatletter
\DeclareRobustCommand{\figref}[1]{\ref@filelist{fig}{#1}}
\makeatother

% Table convenience: \ctable[placement]{filename}{caption}
% The optional placement defaults to [t] (top of the page) and the caption is
% placed above the table.
% The label is auto-derived from the filename without extension, e.g.
% \ctable{fine-tuning/main-table-Qwen3.5-9B.tex}{...} -> \label{tab:main-table-Qwen3.5-9B}
% The file is expected to hold a bare tabular (e.g. pandas to_latex output)
% and is loaded from the repo's code/figures/ dir (requires the xstring package)
\newcommand{\ctable}[3][t]{%
   \stripext{#2}{\ctablabel}%
   \begin{table}[#1]%
      \centering%
      \caption{#3}%
      \label{tab:\ctablabel}%
      \small%
      \input{../../code/figures/#2}%
   \end{table}%
}

% Table reference convenience: \tabref{filename} -> "Table N"
% A comma list works too: \tabref{a,b} -> "Tables 3.1 and 3.2".
% Uses the same filename-derived label as \ctable, e.g.
% \ctable{fine-tuning/main-table-Qwen3.5-9B.tex}{...} then \tabref{fine-tuning/main-table-Qwen3.5-9B.tex}
\makeatletter
\DeclareRobustCommand{\tabref}[1]{\ref@filelist{tab}{#1}}
\makeatother

% --- Automatic reference names ---
% \ref takes its leading word from the label's prefix, so the prose never
% hand-writes (or forgets) "Section", "Chapter", etc.:
%   \ref{chap:background} -> Chapter 2    \ref{sec:values}    -> Section 2.1
%   \ref{subsec:power}    -> Subsection 5.3.2
%   \ref{app:fine-tuning} -> Appendix A   \ref{fig:foo}       -> Figure 4.2
%   \ref{tab:foo}         -> Table 3.1    \ref{eq:nll}        -> Equation 3.4
% A comma-separated list gives the plural word and joins the numbers with
% "and": \ref{sec:a,sec:b} -> Sections 3.2 and 3.3,
% \ref{sec:a,sec:b,sec:c} -> Sections 3.2, 3.3 and 3.4. The word comes from
% the first label, so one list should hold one kind of target.
% Labels with any other (or no) prefix give the plain number; \plainref{...}
% always gives the plain number. To add a prefix, add one line to \ref@word.
\makeatletter
\newif\ifref@plural
\newif\ifref@worded
\newcommand*{\ref@word}[1]{%
   \begingroup
   \def\ref@w{}%
   \IfBeginWith{#1}{chap:}{\def\ref@w{\ifref@plural Chapters\else Chapter\fi}}{}%
   \IfBeginWith{#1}{subsubsec:}{\def\ref@w{\ifref@plural Subsubsections\else Subsubsection\fi}}{}%
   \IfBeginWith{#1}{subsec:}{\def\ref@w{\ifref@plural Subsections\else Subsection\fi}}{}%
   \IfBeginWith{#1}{sec:}{\def\ref@w{\ifref@plural Sections\else Section\fi}}{}%
   \IfBeginWith{#1}{app:}{\def\ref@w{\ifref@plural Appendices\else Appendix\fi}}{}%
   \IfBeginWith{#1}{fig:}{\def\ref@w{\ifref@plural Figures\else Figure\fi}}{}%
   \IfBeginWith{#1}{tab:}{\def\ref@w{\ifref@plural Tables\else Table\fi}}{}%
   \IfBeginWith{#1}{eq:}{\def\ref@w{\ifref@plural Equations\else Equation\fi}}{}%
   \ifx\ref@w\@empty\else\ref@w~\fi
   \endgroup
}
\ExplSyntaxOn
\clist_new:N \l__ref_labels_clist
\int_new:N \l__ref_i_int
\int_new:N \l__ref_n_int
% Print a possibly comma-separated label list as one reference.
\NewDocumentCommand \ref@list { m }
  {
    \clist_set:Nn \l__ref_labels_clist {#1}
    \int_set:Nn \l__ref_n_int { \clist_count:N \l__ref_labels_clist }
    \int_compare:nNnT { \l__ref_n_int } > { 1 } { \ref@pluraltrue }
    \int_zero:N \l__ref_i_int
    \clist_map_inline:Nn \l__ref_labels_clist
      {
        \int_incr:N \l__ref_i_int
        \int_compare:nNnT { \l__ref_i_int } = { 1 }
          { \ifref@worded \ref@word{##1} \fi }
        \int_compare:nNnT { \l__ref_i_int } > { 1 }
          {
            \int_compare:nNnTF { \l__ref_i_int } = { \l__ref_n_int }
              { ~and~ } { ,~ }
          }
        \plainref{##1}
      }
  }
% Map a comma list of figure/table filenames to its labels and reference them:
% {a.png,b} -> \ref{fig:a,fig:b}. \c_colon_str keeps the ":" at catcode 12,
% which is what the prefix tests in \ref@word expect.
\seq_new:N \l__ref_labels_seq
\NewDocumentCommand \ref@filelist { m m }
  {
    \seq_clear:N \l__ref_labels_seq
    \clist_map_inline:nn {#2}
      {
        \stripext{##1}{\ref@filelabel}
        \seq_put_right:Nx \l__ref_labels_seq { #1 \c_colon_str \ref@filelabel }
      }
    \exp_args:Nx \ref { \seq_use:Nn \l__ref_labels_seq {,} }
  }
\ExplSyntaxOff
% The kernel defines \ref as an ltcmd command, so a plain \let would leave
% \plainref pointing at the code we are about to replace (endless recursion):
% \NewCommandCopy makes a real copy of the current \ref.
\AtBeginDocument{%
   \NewCommandCopy{\plainref}{\ref}%
   \RenewDocumentCommand{\ref}{s m}{%
      \begingroup
      \ref@pluralfalse
      \IfBooleanTF{#1}{\ref@wordedfalse}{\ref@wordedtrue}%
      \ref@list{#2}%
      \endgroup
   }%
   % amsmath's \eqref only adds the parentheses, so it wants the bare number
   % (otherwise \eqref{eq:x} would read "(Equation 3.4)").
   \DeclareRobustCommand{\eqref}[1]{%
      \textup{(%
         \begingroup\ref@wordedfalse\ref@pluralfalse\ref@list{#1}\endgroup
      )}%
   }%
}
\makeatother

% --- Research question (RQ) system ---
% Define the research questions once (e.g. in the introduction) and then
% cross-reference any of them throughout the document with a clickable link.
% In an electronic PDF the reference shows a tooltip with the full question
% text when hovered; in print only the "RQn" link label is seen. The tooltip
% is a plain PDF Widget annotation (no extra packages required) - it works in
% desktop readers (Okular, Evince, Acrobat); some viewers ignore it.
%
% Usage (the label string needs no "rq:" prefix - it is added automatically):
%   \newrq{label}{Question text ...}  % define & print an RQ (auto-numbered)
%   \rqref{label}                     % "RQn" clickable link (hover for full text)
%   \rqtext{label}                    % the full question text inline in prose
%
% Example:
%   \newrq{align}{Can we ...?}
%   ... see \rqref{align} in Chapter 3.
\newcounter{rqcount}
\newcommand{\newrq}[2]{%
   \stepcounter{rqcount}%
   \expandafter\xdef\csname rqnum@#1\endcsname{RQ\number\csname c@rqcount\endcsname}%
   \expandafter\gdef\csname rqtext@#1\endcsname{#2}%
   \hypertarget{rq:#1}{}%
   \textbf{\csname rqnum@#1\endcsname.}\ #2%
}
% \pdfhover{visible text}{hover text} - a hover tooltip annotation overlaying
% the visible text (PDF Widget button with zero border). Used by the RQ and
% glossary reference macros below. The hover text is fully expanded into
% \tttmp first so \pdfescapestring sees the literal string (safe for
% parentheses/quotes). The dict is fully expanded at call time (hence
% \expanded), so several tooltips on one page each keep their own text.
\newcommand{\pdfhover}[2]{%
   \edef\tttmp{#2}%
   \leavevmode
   \setbox0=\hbox{#1}%
   \expanded{\noexpand\pdfannot width \the\wd0 height \the\ht0 depth \the\dp0 %
      {/Subtype /Widget /FT /Btn /Ff 65536 /F 768 /H /N /BS << /W 0 >>
       /TU (\pdfescapestring{\tttmp}) /Contents (\pdfescapestring{\tttmp})}}%
   #1%
}
% \rqref and \termref use \texorpdfstring so bookmarks/PDF outlines get plain
% text, and \protect so the links survive writes (e.g. \section headings and
% \caption arguments going into the TOC/LOF): the write keeps the tokens and
% the link is typeset when the list is rendered.
\newcommand{\rqref}[1]{%
   \texorpdfstring{\protect\hyperlink{rq:#1}{\protect\pdfhover{\csname rqnum@#1\endcsname}{\csname rqtext@#1\endcsname}}}{\csname rqnum@#1\endcsname}%
}
\newcommand{\rqtext}[1]{\csname rqtext@#1\endcsname}

% --- Glossary of terms system ---
% Define terms once (e.g. in a "Glossary of terms" section in the
% introduction) and then reference them throughout the document with a
% clickable link. In an electronic PDF the reference shows the definition as
% a tooltip when hovered (same mechanism as the RQ references); in print only
% the linked term text is seen.
%
% Usage (labels need no prefix - internal namespacing is automatic):
%   \newterm{label}{Singular}{Plural}{Definition text}  % register a term
%   \term{label}                 % bold clickable term (hover tooltip)
%   \termref{label}              % clickable term link, not bold
%   \termtext{label}             % the definition text inline in prose
%   \printglossary               % prints the entries only: wrap it in your
%                                % own \section / \chapter heading, e.g. an
%                                % appendix chapter.
%
% A trailing "s" directly after the command selects the plural text, so the
% prose never hand-writes a plural:
%   \term{work-profile}s -> Work profiles      \termref{value-survey}s -> Value surveys
% The "s" is only eaten when it directly follows the closing brace; an
% unregistered label falls back to plain bold text (for \term) or plain text
% (for \termref), so ordinary emphasis still works.
%
% Example:
%   \newterm{llm}{Large language model (LLM)}{Large language models (LLMs)}{A neural network trained on ...}
%   ... see \termref{llm} ...
%
% Terms must be registered (\newterm) before their first reference, the same
% restriction as for research questions. In the report the terms live in a
% single file (appendices/glossary.tex) which is \input twice: once in the
% preamble (registers the terms so \termref works anywhere) and once as the
% appendix (prints them; set \glossaryprinttrue first). \newterm is
% idempotent, so the second \input does not register the terms twice.
\newif\ifglossaryprint
\newcommand{\glistall}{}
\newcommand{\newterm}[4]{%
   \ifcsname gldef-#1\endcsname
   \else
      \expandafter\gdef\csname gltxt-#1\endcsname{#2}%
      \expandafter\gdef\csname glpl-#1\endcsname{#3}%
      \expandafter\gdef\csname gldef-#1\endcsname{#4}%
      \xdef\glistall{\ifx\glistall\empty\else\glistall,\fi#1}%
   \fi
}
\makeatletter
% The term text as a clickable link; the definition is a hover tooltip in an
% electronic PDF and the plain text is used in bookmarks.
\newcommand{\gl@link}[2]{%
   \texorpdfstring{\protect\hyperlink{gl:#1}{\protect\pdfhover{#2}{\csname gldef-#1\endcsname}}}{#2}%
}
% #1 label, #2 1 for bold, #3 the stored text (gltxt or glpl)
\newcommand{\gl@put}[3]{%
   \ifnum#2=\@ne \textbf{\gl@link{#1}{#3}}\else\gl@link{#1}{#3}\fi
}
% Remember the label and weight, then peek at the very next token (no space
% skipping, so a space before a word starting with "s" is safe).
\newcommand{\gl@look}[2]{% #1 label, #2 1 for bold
   \def\gl@tlab{#1}\def\gl@tbold{#2}%
   \futurelet\@let@token\gl@pick
}
% A directly following "s" selects the plural form and is eaten; otherwise
% the singular is used.
\newcommand{\gl@pick}{%
   \ifx\@let@token s%
      \expandafter\@firstoftwo
   \else
      \expandafter\@secondoftwo
   \fi
   {\gl@put{\gl@tlab}{\gl@tbold}{\csname glpl-\gl@tlab\endcsname}\@gobble}%
   {\gl@put{\gl@tlab}{\gl@tbold}{\csname gltxt-\gl@tlab\endcsname}}%
}
\DeclareRobustCommand{\term}[1]{%
   \ifcsname gltxt-#1\endcsname
      \expandafter\@firstoftwo
   \else
      \expandafter\@secondoftwo
   \fi
   {\gl@look{#1}{\@ne}}%
   {\textbf{#1}}%
}
\DeclareRobustCommand{\termref}[1]{%
   \ifcsname gltxt-#1\endcsname
      \expandafter\@firstoftwo
   \else
      \expandafter\@secondoftwo
   \fi
   {\gl@look{#1}{\z@}}%
   {#1}%
}
\newcommand{\termtext}[1]{\csname gldef-#1\endcsname}
\newcommand{\printglossary}{%
   \expandafter\gl@printlist\expandafter{\glistall}%
}
\newcommand{\gl@printlist}[1]{%
   \@for\gl@lab:=#1\do{%
      \par\noindent
      \hypertarget{gl:\gl@lab}{}\textbf{\csname gltxt-\gl@lab\endcsname:}\ %
      \csname gldef-\gl@lab\endcsname\par\vspace{6pt}%
   }%
}
\makeatother

% --- Model glossary system ---
% The same idea as the term glossary, but for models. A reference is set in
% \texttt, links to the model's entry in the appendix, and shows a short
% description as a hover tooltip in an electronic PDF. Each entry also holds
% the model's Hugging Face (or provider) URL, printed with the appendix entry.
%
% Usage (labels need no prefix - internal namespacing is automatic):
%   \newmodel{label}{Display name}{Short description}{URL}
%   \modelref{label}             % linked \texttt{Display name} (hover tooltip)
%   \modeltext{label}            % the short description inline in prose
%   \modelurl{label}             % the URL alone
%   \printmodels                 % prints the entries only: wrap it in your
%                                % own \section / \chapter heading, e.g. an
%                                % appendix chapter.
%
% Example:
%   \newmodel{qwen}{Qwen3.8-27B}{Open-weight ...}{https://huggingface.co/...}
%   ... see \modelref{qwen} ...
%
% Models must be registered (\newmodel) before their first reference, as with
% terms. In the report they live in appendices/models.tex, which is \input
% twice: once in the preamble (registration) and once as the appendix (print;
% set \modelprinttrue first). \newmodel is idempotent, so the second input
% does not register the models twice.
\newif\ifmodelprint
\newcommand{\gmlistall}{}
\newcommand{\newmodel}[4]{%
   \ifcsname gmdef-#1\endcsname
   \else
      \expandafter\gdef\csname gmtxt-#1\endcsname{#2}%
      \expandafter\gdef\csname gmdef-#1\endcsname{#3}%
      \expandafter\gdef\csname gmurl-#1\endcsname{#4}%
      \xdef\gmlistall{\ifx\gmlistall\empty\else\gmlistall,\fi#1}%
   \fi
}
\newcommand{\modelref}[1]{%
   \texorpdfstring{\protect\hyperlink{gm:#1}{\protect\pdfhover{\texttt{\csname gmtxt-#1\endcsname}}{\csname gmdef-#1\endcsname}}}{\csname gmtxt-#1\endcsname}%
}
\newcommand{\modeltext}[1]{\csname gmdef-#1\endcsname}
\newcommand{\modelurl}[1]{\csname gmurl-#1\endcsname}
\makeatletter
\newcommand{\printmodels}{%
   \expandafter\gm@printlist\expandafter{\gmlistall}%
}
\newcommand{\gm@printlist}[1]{%
   \@for\gm@lab:=#1\do{%
      \par\noindent
      \hypertarget{gm:\gm@lab}{}\texttt{\csname gmtxt-\gm@lab\endcsname}: %
      \csname gmdef-\gm@lab\endcsname\par
      \nopagebreak\noindent
      \href{\csname gmurl-\gm@lab\endcsname}{\footnotesize\url{\csname gmurl-\gm@lab\endcsname}}%
      \par\vspace{6pt}%
   }%
}
\makeatother
